set(SRCS imx8_boot.c imx8_flexspi_nor_boot.c imx8_flexspi_nor_flash.c)

  list(APPEND SRCS imx8_appinit.c imx8_bringup.c)

  list(APPEND SRCS imx8_userleds.c)

  list(APPEND SRCS imx8_buttons.c)

endif()

if(CONFIG_IMX8_LPSPI)

endif()

if(CONFIG_IMX8_LCD)
  list(APPEND SRCS imx8_lcd.c)
endif()

  list(APPEND SRCS imx8_mmcsd_spi.c)
endif()

if(CONFIG_DEV_GPIO)
  list(APPEND SRCS imx8_gpio.c)
